```dart
/// Pumps frames until [condition] holds or [timeout] elapses.
Future<void> pumpUntil(
  WidgetTester tester,
  bool Function() condition, {
  required String description,
  Duration timeout = const Duration(seconds: 10),
}) async {
  final end = DateTime.now().add(timeout);
  while (DateTime.now().isBefore(end)) {
    await tester.pump(const Duration(milliseconds: 100));
    if (condition()) return;
  }
  fail('Timed out waiting for $description');
}

/// Pumps frames until [finder] matches or [timeout] elapses.
Future<void> pumpUntilFound(
  WidgetTester tester,
  Finder finder, {
  Duration timeout = const Duration(seconds: 10),
}) => pumpUntil(
  tester,
  () => finder.evaluate().isNotEmpty,
  description: '$finder',
  timeout: timeout,
);

/// Localizations of the currently shown screen.
AppLocalizations l10nOf(WidgetTester tester) {
  final l10n = AppLocalizations.of(tester.element(find.byType(Scaffold).first));
  if (l10n == null) fail('AppLocalizations not available');
  return l10n;
}

/// Starts the app with an in-memory preference store seeded with
/// [preferences]. The real app's preferences on the device stay untouched.
Future<void> launchApp(
  WidgetTester tester, {
  Map<String, Object> preferences = const <String, Object>{},
}) async {
  SharedPreferences.setMockInitialValues(preferences);
  await tester.pumpWidget(const BreathscapeApp());
}

/// Starts the app fresh and accepts the disclaimer, leaving the session page
/// ready.
Future<void> launchToSessionPage(WidgetTester tester) async {
  await launchApp(tester);
  await pumpUntilFound(tester, find.byType(DisclaimerPage));

  await tester.tap(find.text(l10nOf(tester).disclaimerAcceptButton));
  await pumpUntilFound(tester, find.byType(BreathingAnimationWidget));
}
```

| Funktion | vorher (`breathing_session_test.dart`) | Verhalten |
|----------|----------------------------------------|-----------|
| `pumpUntil` | – (neu) | Pumpt in 100-ms-Schritten, bis eine beliebige Bedingung `true` liefert; nach dem Timeout (Standard 10 s) folgt `fail(...)` mit `description`. |
| `pumpUntilFound` | `_pumpUntilFound` | Delegiert jetzt an `pumpUntil` mit der Bedingung `finder.evaluate().isNotEmpty`. |
| `l10nOf` | `_l10n` | Unverändert; liefert `AppLocalizations` über den ersten `Scaffold`, bei `null` folgt `fail(...)` (kein `!`). |
| `launchApp` | – (neu, vorher Teil von `_launchToSessionPage`) | Setzt `SharedPreferences.setMockInitialValues(preferences)` und pumpt `BreathscapeApp`. Über `preferences` kann der In-Memory-Store vorbelegt werden (Standard: leer). |
| `launchToSessionPage` | `_launchToSessionPage` | Ruft `launchApp` ohne Vorbelegung auf, wartet auf `DisclaimerPage`, tippt auf `disclaimerAcceptButton` und wartet auf `BreathingAnimationWidget`. |

**Umgang mit SharedPreferences:** Alle Integrationstests laufen gegen den In-Memory-Store von `SharedPreferences.setMockInitialValues`. Die echten Präferenzen auf dem Gerät werden nicht gelesen oder geschrieben. `DisclaimerRepository` (`lib/features/disclaimer/domain/disclaimer_repository.dart`) liest und schreibt über `SharedPreferences.getInstance()` und damit in diesen Store.

**Umgang mit Lokalisierung:** `BreathscapeApp` setzt keine feste `locale`; die Sprache folgt dem Gerät (`app_en.arb` / `app_de.arb`). Die Integrationstests vergleichen deshalb nicht mit Literalen, sondern mit Werten aus `l10nOf(tester)` (`disclaimerAcceptButton`, `sessionReady`, `settingsTitle`, `voiceCuesOn`, `voiceCuesOff`).
